  'Client ID', 'Client Name', 'Country', 'Status', 'Created Date'

  'Campaign Name', // col 2  (= projectName)
  'Client ID',     // col 3
  'Client Name',   // col 4
  'Project ID',    // col 5

/**
 * S92.F01 — Find an existing client by name (case-insensitive) in _Clients,
 * or create a new row and return the new Client ID.
 *
 * @param  {string} clientName
 * @param  {string} country
 * @returns {string} clientId (e.g. "CLI-20260616-0001") or '' if name blank
 */
function findOrCreateClient_(clientName, country) {
  var ss    = getMasterSS_();
  var sheet = ss.getSheetByName('_Clients') ||
              ensureSheet_(ss, '_Clients', CLIENT_HEADERS_V292_, '#1a3650');

  var cNameLc = String(clientName || '').trim().toLowerCase();
  if (!cNameLc) return '';

  // Scan existing rows
  if (sheet.getLastRow() >= 2) {
    var data = sheet.getRange(2, 1, sheet.getLastRow() - 1, 2).getValues();
    for (var i = 0; i < data.length; i++) {
      if (String(data[i][1]).trim().toLowerCase() === cNameLc) {
        return String(data[i][0]).trim(); // found — return existing ID
      }
    }
  }

  // Not found — create
  var props   = PropertiesService.getScriptProperties();
  var counter = parseInt(props.getProperty('kai_client_id_counter') || '0') + 1;
  props.setProperty('kai_client_id_counter', String(counter));
  var clientId = 'CLI-' + Utilities.formatDate(new Date(), 'Asia/Kolkata', 'yyyyMMdd') +
                 '-' + String(counter).padStart(4, '0');
  var today    = Utilities.formatDate(new Date(), 'Asia/Kolkata', 'dd-MMM-yyyy');

  sheet.appendRow([
    clientId,
    String(clientName || '').trim(),
    String(country    || '').trim(),
    'ACTIVE',
    today
  ]);
  SpreadsheetApp.flush();
  return clientId;
}

/**
 * S92.F02 — Find an existing ACTIVE project by clientName + projectName
 * (case-insensitive) using existing listProjects(), or create a new one
 * via existing createProject_().
 *
 * @param  {string} clientName
 * @param  {string} projectName
 * @param  {string} country
 * @returns {string} projectId (e.g. "PROJ-20260616-001")
 */
function findOrCreateProject_(clientName, projectName, country) {
  var pNameLc = String(projectName || '').trim().toLowerCase();
  var cNameLc = String(clientName  || '').trim().toLowerCase();

  var projects = listProjects();
  for (var i = 0; i < projects.length; i++) {
    var p = projects[i];
    if (
      p.name.toLowerCase()   === pNameLc &&
      p.client.toLowerCase() === cNameLc &&
      p.status.toUpperCase() === 'ACTIVE'
    ) {
      return p.id; // reuse existing
    }
  }

  // Not found — delegate to V284 createProject_
  return createProject_(projectName, clientName, country);
}

/**
 * S92.F03 — Find an existing ACTIVE campaign for a given project, or create
 * a new row in _Campaigns.
 *
 * Match priority:
 *   1. projectId match (most precise)
 *   2. clientName + campaignName match (fallback, case-insensitive)
 *
 * @param  {string} clientId
 * @param  {string} clientName
 * @param  {string} projectId
 * @param  {string} projectName   used as Campaign Name
 * @param  {string} country
 * @returns {{ campaignId, campaignName, campaignCreated }}
 */
function findOrCreateCampaign_(clientId, clientName, projectId, projectName, country) {
  var ss    = getMasterSS_();
  var sheet = ss.getSheetByName('_Campaigns') ||
              ensureSheet_(ss, '_Campaigns', CAMPAIGN_HEADERS_V292_, '#1e3a5f');

  var pNameLc = String(projectName || '').trim().toLowerCase();
  var cNameLc = String(clientName  || '').trim().toLowerCase();

  if (sheet.getLastRow() >= 2) {
    var rows = sheet.getRange(
      2, 1, sheet.getLastRow() - 1, CAMPAIGN_HEADERS_V292_.length
    ).getValues();

    for (var i = 0; i < rows.length; i++) {
      var r      = rows[i];
      var status = String(r[6]).toUpperCase().trim();
      if (status !== 'ACTIVE') continue;

      var matchByProject = projectId && String(r[4]).trim() === projectId;
      var matchByName    = String(r[3]).trim().toLowerCase() === cNameLc &&
                           String(r[1]).trim().toLowerCase() === pNameLc;

      if (matchByProject || matchByName) {
        return {
          campaignId:      String(r[0]).trim(),
          campaignName:    String(r[1]).trim(),
          campaignCreated: false
        };
      }
    }
  }

  // Not found — create
  var props    = PropertiesService.getScriptProperties();
  var counter  = parseInt(props.getProperty('kai_campaign_id_counter') || '0') + 1;
  props.setProperty('kai_campaign_id_counter', String(counter));
  var campaignId   = 'CAMP-' + Utilities.formatDate(new Date(), 'Asia/Kolkata', 'yyyyMMdd') +
                     '-' + String(counter).padStart(4, '0');
  var campaignName = String(projectName || '').trim();
  var today        = Utilities.formatDate(new Date(), 'Asia/Kolkata', 'dd-MMM-yyyy');

  sheet.appendRow([
    campaignId,
    campaignName,
    clientId   || '',
    clientName || '',
    projectId  || '',
    String(country || '').trim(),
    'ACTIVE',
    today,
    0              // Req Count starts at 0
  ]);
  SpreadsheetApp.flush();
  return { campaignId: campaignId, campaignName: campaignName, campaignCreated: true };
}

/**
 * S92.F04 — Fetch a single campaign record from _Campaigns by Campaign ID.
 * Returns the record object or null if not found.
 * Used by the backward-compat path when caller supplies an existingCampaignId.
 */
function getCampaignById_(campaignId) {
  var sheet = getMasterSS_().getSheetByName('_Campaigns');
  if (!sheet || sheet.getLastRow() < 2) return null;

  var rows = sheet.getRange(
    2, 1, sheet.getLastRow() - 1, CAMPAIGN_HEADERS_V292_.length
  ).getValues();

  for (var i = 0; i < rows.length; i++) {
    if (String(rows[i][0]).trim() === String(campaignId).trim()) {
      return {
        campaignId:   String(rows[i][0]).trim(),
        campaignName: String(rows[i][1]).trim(),
        clientId:     String(rows[i][2]).trim(),
        clientName:   String(rows[i][3]).trim(),
        projectId:    String(rows[i][4]).trim()
      };
    }
  }
  return null;
}

/**
 * S92.F05 — Add `count` to the Req Count column (col 9) on a _Campaigns row.
 */
function incrementCampaignReqCount_(campaignId, count) {
  var sheet = getMasterSS_().getSheetByName('_Campaigns');
  if (!sheet || sheet.getLastRow() < 2) return;

  var ids = sheet.getRange(2, 1, sheet.getLastRow() - 1, 1).getValues();
  for (var i = 0; i < ids.length; i++) {
    if (String(ids[i][0]).trim() === String(campaignId).trim()) {
      var cell = sheet.getRange(i + 2, CAMPAIGN_HEADERS_V292_.length); // col 9
      cell.setValue((parseInt(cell.getValue()) || 0) + (count || 1));
      SpreadsheetApp.flush();
      return;
    }
  }
}

/**
 * S92.F06 — Public dashboard endpoint: create requirements from a JD batch,
 * optionally under a new or existing campaign.
 *
 * CALL PATTERNS
 * ─────────────────────────────────────────────────────────────
 * A) New Client / Project (frontend Option B):
 *    bulkCreateRequirementsFromJDs(batchId, indices,
 *      { clientName, projectName, country }, null)
 *
 * B) Existing campaign (frontend Option A / backward compat):
 *    bulkCreateRequirementsFromJDs(batchId, indices, null, 'CAMP-xxx')
 *
 * C) No campaign (backward compat — saves requirements only):
 *    bulkCreateRequirementsFromJDs(batchId, indices)
 *
 * @param {string}   batchId            ScriptCache key (JD_BATCH_*)
 * @param {number[]} indices            Positions to save; [] or omit = save all
 * @param {Object}   newCampaign        { clientName, projectName, country } or null
 * @param {string}   existingCampaignId Existing CAMP-xxx ID or null
 *
 * @returns {{
 *   ok:              boolean,
 *   campaignCreated: boolean,
 *   campaignId:      string,
 *   campaignName:    string,
 *   requirements:    Array<{reqId, trade, clientName, deployCountry, quantity}>,
 *   msg:             string,
 *   errors:          string[]
 * }}
 */
function bulkCreateRequirementsFromJDs(batchId, indices, newCampaign, existingCampaignId) {
  var result = {
    ok:              false,
    campaignCreated: false,
    campaignId:      '',
    campaignName:    '',
    requirements:    [],
    msg:             '',
    errors:          []
  };

  try {
    // ── 1. Validate batchId ─────────────────────────────────────────
    if (!batchId) {
      result.msg = 'batchId is required.';
      return result;
    }

    // ── 2. Retrieve JDs from ScriptCache ───────────────────────────
    var cached = CacheService.getScriptCache().get('JD_BATCH_' + batchId);
    if (!cached) {
      result.msg = 'Batch data expired or not found. Please re-upload the JD file.';
      return result;
    }

    var allJds  = JSON.parse(cached);
    var toSave  = [];
    var idxList = indices || [];

    if (idxList.length === 0) {
      toSave = allJds;                            // save everything if no filter
    } else {
      idxList.forEach(function(idx) {
        if (idx >= 0 && idx < allJds.length) toSave.push(allJds[idx]);
      });
    }

    if (toSave.length === 0) {
      result.msg = 'No valid JD indices found in batch.';
      return result;
    }

    // ── 3. Resolve campaign ─────────────────────────────────────────
    var campaignId      = '';
    var campaignName    = '';
    var campaignCreated = false;
    var projectId       = '';

    if (existingCampaignId) {
      // ── Pattern B: caller supplies a known Campaign ID ───────────
      var existingCamp = getCampaignById_(existingCampaignId);
      if (!existingCamp) {
        result.msg = 'Campaign not found: ' + existingCampaignId;
        return result;
      }
      campaignId   = existingCamp.campaignId;
      campaignName = existingCamp.campaignName;
      projectId    = existingCamp.projectId;
      // campaignCreated stays false

    } else if (newCampaign && newCampaign.projectName) {
      // ── Pattern A: create/reuse full hierarchy ───────────────
      var clientName  = String(newCampaign.clientName  || '').trim();
      var pName       = String(newCampaign.projectName || '').trim();
      var country     = String(newCampaign.country     || '').trim();

      var clientId = findOrCreateClient_(clientName, country);
      projectId    = findOrCreateProject_(clientName, pName, country);
      var camp     = findOrCreateCampaign_(clientId, clientName, projectId, pName, country);

      campaignId      = camp.campaignId;
      campaignName    = camp.campaignName;
      campaignCreated = camp.campaignCreated;
    }
    // Pattern C: no campaign params — fall through with empty campaignId

    result.campaignId      = campaignId;
    result.campaignName    = campaignName;
    result.campaignCreated = campaignCreated;

    // ── 4. Save each JD as a Requirement ─────────────────────────
    var requirements = [];
    var saved = 0;

    toSave.forEach(function(jd) {
      try {
        // Stamp campaign fields onto JD only when not already set by extractor
        if (newCampaign) {
          if (!jd.clientName    && newCampaign.clientName) jd.clientName    = newCampaign.clientName;
          if (!jd.deployCountry && newCampaign.country)    jd.deployCountry = newCampaign.country;
        }

        var reqId = saveRequirementV284_(jd);

        // Map requirement to project in _Requirements Notes column
        if (projectId) {
          mapReqToProject_(reqId, projectId);
        }

        requirements.push({
          reqId:         reqId,
          trade:         jd.trade         || '',
          department:    jd.department    || '',
          clientName:    jd.clientName    || '',
          deployCountry: jd.deployCountry || '',
          quantity:      jd.quantity      || 1
        });
        saved++;
      } catch(saveErr) {
        result.errors.push('JD "' + (jd.trade || 'unknown') + '": ' + saveErr.message);
        Logger.log('S92.F06 save error: ' + saveErr.message);
      }
    });

    // ── 5. Update campaign Req Count ────────────────────────────
    if (campaignId && saved > 0) {
      try { incrementCampaignReqCount_(campaignId, saved); } catch(e) {}
    }

    // ── 6. Build response ─────────────────────────────────────
    result.ok           = saved > 0;
    result.requirements = requirements;
    result.msg = saved + ' requirement' + (saved !== 1 ? 's' : '') + ' saved' +
                 (campaignId
                   ? ' under ' + (campaignCreated ? 'new' : 'existing') +
                     ' campaign ' + campaignId
                   : '') +
                 (result.errors.length > 0
                   ? ' (' + result.errors.length + ' error' +
                     (result.errors.length !== 1 ? 's' : '') + ')'
                   : '');
    return result;

  } catch(err) {
    result.msg = 'bulkCreateRequirementsFromJDs error: ' + err.message;
    Logger.log('S92.F06 fatal: ' + err.message);
    return result;
  }
}
