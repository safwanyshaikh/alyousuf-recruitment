// Drop this file into the SAME Apps Script project as Code.gs, then run
// setupWeeklyBdReport() ONCE from the editor to install the trigger.
//
// What it does:
//   • Fires every Sunday 07:00–07:59 IST (project timezone = Asia/Kolkata)
//     so the report lands in the inbox BEFORE 08:00.
//   • Researches overseas + emerging-corridor hiring via grounded Gemini
//     (Google Search tool enabled) and emails ONE plain-text, action-only
//     report — no icons, no tables.
//   • Sends FROM the account that owns/authorizes this script
//     (safwanyshaikh@gmail.com) TO bd@alyousufent.com, CC ayeadmin@alyousufent.com.
//
// Reuses existing Code.gs helpers when present: validateApiKey_, appendLog_,
// sendAlert_. Falls back gracefully if any are missing.
// ═══════════════════════════════════════════════════════════════════

var BD_REPORT_CONFIG = {
  to:              'bd@alyousufent.com',
  cc:              'ayeadmin@alyousufent.com',
  fromName:        'AL Yousuf Enterprises LLP — BD Intelligence',
  triggerHour:     7,                 // Sunday 07:00–07:59 IST → lands before 08:00
  timeZone:        'Asia/Kolkata',
  model:           'gemini-2.5-flash',
  endpoint:        'https://generativelanguage.googleapis.com/v1beta/models/',
  maxOutputTokens: 8000,
  retries:         5
};

// ───────────────────────────────────────────────────────────────────
// SETUP — run once from the editor
// ───────────────────────────────────────────────────────────────────
function setupWeeklyBdReport() {
  removeWeeklyBdReport();
  ScriptApp.newTrigger('sendWeeklyBdReport')
    .timeBased()
    .onWeekDay(ScriptApp.WeekDay.SUNDAY)
    .atHour(BD_REPORT_CONFIG.triggerHour)
    .create();
  Logger.log('Weekly BD report trigger installed: Sunday ' +
             BD_REPORT_CONFIG.triggerHour + ':00 ' + BD_REPORT_CONFIG.timeZone);
}

function removeWeeklyBdReport() {
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (t.getHandlerFunction() === 'sendWeeklyBdReport') ScriptApp.deleteTrigger(t);
  });
}

// Build + send immediately (manual test; does NOT wait for Sunday).
function runWeeklyBdReportNow() {
  sendWeeklyBdReport();
}

// ───────────────────────────────────────────────────────────────────
// ENTRY POINT — Sunday trigger
// ───────────────────────────────────────────────────────────────────
function sendWeeklyBdReport() {
  try {
    var report = callGeminiBdText_(buildBdReportPrompt_());
    if (!report || report.length < 60) {
      bdLog_('BD_REPORT_FAILED', 'Gemini returned empty/short output');
      bdAlert_('Weekly BD Report', 'Gemini returned empty output — report not sent.');
      return;
    }
    var weekOf = Utilities.formatDate(new Date(), BD_REPORT_CONFIG.timeZone, 'dd MMM yyyy');
    GmailApp.sendEmail(
      BD_REPORT_CONFIG.to,
      'KAI Weekly BD Report — Week of ' + weekOf,
      report,
      { cc: BD_REPORT_CONFIG.cc, name: BD_REPORT_CONFIG.fromName }
    );
    bdLog_('BD_REPORT_SENT', 'to=' + BD_REPORT_CONFIG.to + ' cc=' + BD_REPORT_CONFIG.cc + ' week=' + weekOf);
  } catch (e) {
    bdLog_('BD_REPORT_FAILED', e.message);
    bdAlert_('Weekly BD Report failed', String(e.message));
  }
}

// ───────────────────────────────────────────────────────────────────
// GEMINI — grounded plain-text call (Google Search tool enabled)
// ───────────────────────────────────────────────────────────────────
function callGeminiBdText_(promptText) {
  var apiKey = (typeof validateApiKey_ === 'function')
    ? validateApiKey_()
    : PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY');
  if (!apiKey) throw new Error('GEMINI_API_KEY not configured in Script Properties.');

  var url = BD_REPORT_CONFIG.endpoint + BD_REPORT_CONFIG.model +
            ':generateContent?key=' + encodeURIComponent(apiKey);

  var payload = {
    contents: [{ role: 'user', parts: [{ text: promptText }] }],
    tools: [{ google_search: {} }],   // live grounding for verification
    generationConfig: {
      temperature: 0.3,
      topP: 0.9,
      maxOutputTokens: BD_REPORT_CONFIG.maxOutputTokens
    }
  };

  var lastErr = null;
  for (var attempt = 1; attempt <= BD_REPORT_CONFIG.retries; attempt++) {
    try {
      Utilities.sleep(attempt > 1 ? Math.min(Math.pow(2, attempt - 1) * 1500, 30000) : 0);
      var resp = UrlFetchApp.fetch(url, {
        method: 'post', contentType: 'application/json',
        payload: JSON.stringify(payload), muteHttpExceptions: true
      });
      var code = resp.getResponseCode();
      var text = resp.getContentText();
      if (code === 200) return bdExtractText_(text);
      lastErr = new Error('Gemini HTTP ' + code + ': ' + text.slice(0, 300));
    } catch (e) {
      lastErr = e;
    }
  }
  throw lastErr || new Error('BD Gemini call failed after ' + BD_REPORT_CONFIG.retries + ' attempts.');
}

// Concatenate all text parts from the Gemini response envelope.
function bdExtractText_(rawJson) {
  try {
    var obj   = (typeof rawJson === 'string') ? JSON.parse(rawJson) : rawJson;
    var parts = obj.candidates[0].content.parts || [];
    return parts.map(function (p) { return p.text || ''; }).join('').trim();
  } catch (e) {
    return '';
  }
}

// ───────────────────────────────────────────────────────────────────
// Logging / alert shims — use Code.gs helpers if present, else fall back
// ───────────────────────────────────────────────────────────────────
function bdLog_(status, notes) {
  try {
    if (typeof appendLog_ === 'function') { appendLog_({ status: status, notes: notes }); return; }
  } catch (e) {}
  Logger.log(status + ': ' + notes);
}

function bdAlert_(subject, body) {
  try {
    if (typeof sendAlert_ === 'function') { sendAlert_(subject, body); return; }
  } catch (e) {}
  Logger.log('[KAI Alert] ' + subject + ': ' + body);
}
